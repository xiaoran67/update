def read_txt_to_array(file_name):
    """读取文本文件到数组"""
    try:
        with open(file_name, 'r', encoding='utf-8') as file:
            return [line.strip() for line in file if line.strip()]
    except FileNotFoundError:
        print(f"⚠️ 文件未找到: {file_name}")
        return []
    except Exception as e:
        print(f"❌ 读取文件错误 {file_name}: {e}")
        return []

def traditional_to_simplified(text: str) -> str:
    """繁体转简体"""
    try:
        converter = opencc.OpenCC('t2s')
        return converter.convert(text)
    except Exception as e:
        print(f"❌ 繁简转换错误: {e}")
        return text

def read_blacklist_from_txt(file_path):
    """读取黑名单"""
    try:
        with open(file_path, 'r', encoding='utf-8') as file:
            return [line.split(',')[1].strip() for line in file if ',' in line]
    except Exception as e:

        return []

def get_url_hash(url):
    """获取URL的哈希值用于去重"""
    return hashlib.md5(url.encode('utf-8')).hexdigest()

def should_skip_url(url):

def clean_channel_name(channel_name, removal_list):
    """清理频道名称中的特定字符"""
    for item in removal_list:
        channel_name = channel_name.replace(item, "")
    
    if channel_name.endswith("HD"):
        channel_name = channel_name[:-2]
    if channel_name.endswith("台") and len(channel_name) > 3:
        channel_name = channel_name[:-1]
    
    return channel_name.strip()

def process_name_string(input_str):
    """处理频道名称字符串"""
    try:
        parts = input_str.split(',')
        processed_parts = []
        
        for part in parts:
            if "CCTV" in part and "://" not in part:
                part = part.replace("IPV6", "").replace("PLUS", "+").replace("1080", "")
                filtered_str = ''.join(char for char in part if char.isdigit() or char in 'K+')
                
                if not filtered_str.strip():
                    filtered_str = part.replace("CCTV", "")
                
                if len(filtered_str) > 2 and re.search(r'4K|8K', filtered_str):
                    filtered_str = re.sub(r'(4K|8K).*', r'\1', filtered_str)
                    if len(filtered_str) > 2: 
                        filtered_str = re.sub(r'(4K|8K)', r'(\1)', filtered_str)
                
                processed_parts.append("CCTV" + filtered_str)
            elif "卫视" in part:
                processed_parts.append(re.sub(r'卫视「.*」', '卫视', part))
            else:
                processed_parts.append(part)
        
        return ','.join(processed_parts)
    except Exception as e:
        print(f"❌ 处理频道名称错误: {e}, 输入: {input_str}")
        return input_str

def clean_url(url):
    """清理URL中的$符号及之后内容"""
    try:
        last_dollar_index = url.rfind('$')
        return url[:last_dollar_index] if last_dollar_index != -1 else url
    except Exception as e:
        print(f"❌ 清理URL错误: {e}")
        return url

def convert_m3u_to_txt(m3u_content):
    """M3U格式转TXT格式"""
    try:
        lines = m3u_content.split('\n')
        txt_lines = []
        channel_name = ""
        
        for line in lines:
            if line.startswith("#EXTM3U"):
                continue
            elif line.startswith("#EXTINF"):
                channel_name = line.split(',')[-1].strip()
            elif line.startswith(("http", "rtmp", "p3p")):
                if channel_name:
                    txt_lines.append(f"{channel_name},{line.strip()}")
            
            if "#genre#" not in line and "," in line and "://" in line:
                pattern = r'^[^,]+,[^\s]+://[^\s]+$'
                if bool(re.match(pattern, line)):
                    txt_lines.append(line)
        
        return '\n'.join(txt_lines)
    except Exception as e:
        print(f"❌ 转换M3U到TXT错误: {e}")
        return m3u_content

def get_http_response(url, timeout=None, retries=None, backoff_factor=None):
    """带重试的HTTP请求"""
    timeout = timeout or CONFIG['request_timeout']
    retries = retries or CONFIG['request_retries']
    backoff_factor = backoff_factor or CONFIG['request_backoff_factor']
    
    headers = {
        'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
    }
    
    for attempt in range(retries):
        try:
            req = urllib.request.Request(url, headers=headers)
            with urllib.request.urlopen(req, timeout=timeout) as response:
                data = response.read()
                return data.decode('utf-8')
        except urllib.error.HTTPError as e:
            print(f"❌ [HTTP错误] 代码: {e.code}, URL: {url}")
            break
        except urllib.error.URLError as e:
            print(f"⚠️ [URL错误] 原因: {e.reason}, 尝试: {attempt + 1}/{retries}")
        except socket.timeout:
            print(f"⏰ [超时] URL: {url}, 尝试: {attempt + 1}/{retries}")
        except Exception as e:
            print(f"⚠️ [异常] {type(e).__name__}: {e}, 尝试: {attempt + 1}/{retries}")
        
        if attempt < retries - 1:
            sleep_time = backoff_factor * (2 ** attempt)
            print(f"⏳ 等待 {sleep_time} 秒后重试...")
            time.sleep(sleep_time)
    
    return None

    def __init__(self):

    def discover_local_channels(self):

        try:
            for filename in os.listdir(CONFIG['local_channels_dir']):
                if filename.endswith('.txt'):
                    name = filename[:-4]  # 去掉.txt后缀
                    key = name.replace('频道', '').lower()  # 生成key
                    self.local_categories[key] = {
                        'name': name,
                        'dict_file': filename,
                        'output_name': f"☘️{name}",
                        'lines': [],
                        'url_set': set(),
                        'manual_lines': []  # 手工区频道
                    }
                    print(f"  📁 发现地方台: {name}")
        except Exception as e:
            print(f"❌ 自动发现地方台错误: {e}")
    
    def discover_manual_channels(self):
        """自动发现手工区分类"""
        try:
            for filename in os.listdir(CONFIG['manual_dir']):
                if filename.endswith('.txt') and filename not in ['about.txt', '今日推台.txt', '今日推荐.txt']:
                    name = filename[:-4]  # 去掉.txt后缀
                    key = name.replace('频道', '').lower()  # 生成key
                    self.manual_categories[key] = {
                        'name': name,
                        'file': filename,
                        'output_name': f"⭐{name}",
                        'lines': [],
                        'url_set': set()
                    }
                    print(f"  📁 发现手工区: {name}")
        except Exception as e:
            print(f"❌ 自动发现手工区错误: {e}")
    
    def load_all_dictionaries(self):
        """加载所有字典"""
        # 加载主频道字典
        for key, category in self.main_categories.items():
            dict_path = f"{CONFIG['main_channels_dir']}/{category['dict_file']}"
            category['dictionary'] = read_txt_to_array(dict_path)
            print(f"  📖 加载字典: {category['name']} ({len(category['dictionary'])} 条)")
        
        # 加载地方台字典
        for key, category in self.local_categories.items():
            dict_path = f"{CONFIG['local_channels_dir']}/{category['dict_file']}"
            category['dictionary'] = read_txt_to_array(dict_path)
            print(f"  📖 加载字典: {category['name']} ({len(category['dictionary'])} 条)")
    
    def process_channel_line(self, line):
        """处理单行频道数据并分类"""
        try:
            if "#genre#" not in line and "#EXTINF:" not in line and "," in line and "://" in line:
                channel_name = line.split(',')[0].strip()
                channel_name = clean_channel_name(channel_name, CONFIG['removal_list'])
                channel_name = traditional_to_simplified(channel_name)
                channel_address = clean_url(line.split(',')[1].strip())
                processed_line = channel_name + "," + channel_address

                # 跳过黑名单和特定协议
                if channel_address in combined_blacklist or should_skip_url(channel_address):
                    return

                url_hash = get_url_hash(channel_address)
                
                # 分发到各个分类 - 独立判断，允许重复
                # 主频道分发
                for key, category in self.main_categories.items():
                    if self.should_add_to_category(channel_name, url_hash, category):
                        category['url_set'].add(url_hash)
                        category['lines'].append(process_name_string(processed_line))
                
                # 地方台分发
                for key, category in self.local_categories.items():
                    if self.should_add_to_category(channel_name, url_hash, category):
                        category['url_set'].add(url_hash)
                        category['lines'].append(process_name_string(processed_line))
                
                # 特殊分类分发
                for key, category in self.special_categories.items():
                    if self.should_add_to_special_category(channel_name, url_hash, category):
                        category['url_set'].add(url_hash)
                        category['lines'].append(process_name_string(processed_line))
                
                # 其他频道分发
                if (url_hash not in others_url_set and 
                    not any(url_hash in cat['url_set'] for cat in self.main_categories.values()) and
                    not any(url_hash in cat['url_set'] for cat in self.local_categories.values()) and
                    not any(url_hash in cat['url_set'] for cat in self.special_categories.values())):
                    others_url_set.add(url_hash)
                    others_lines.append(processed_line)
                    
        except Exception as e:
            print(f"❌ 处理频道行错误: {e}, 行内容: {line}")
    
    def should_add_to_category(self, channel_name, url_hash, category):
        """判断是否应该添加到分类"""
        if url_hash in category['url_set']:
            return False
            
        dictionary = category.get('dictionary', [])
        if not dictionary:
            return False
            
        return channel_name in dictionary
    
    def should_add_to_special_category(self, channel_name, url_hash, category):
        """判断是否应该添加到特殊分类"""
        if url_hash in category['url_set']:
            return False
            
        keywords = category.get('keywords', [])
        if not keywords:
            return False
            
        return any(keyword.lower() in channel_name.lower() for keyword in keywords)
    
    def process_manual_channels(self):
        """处理手工区频道"""
        for key, category in self.manual_categories.items():
            file_path = f"{CONFIG['manual_dir']}/{category['file']}"
            
            if os.path.exists(file_path):
                manual_data = read_txt_to_array(file_path)
                added_count = 0
                
                for line in manual_data:
                    if ',' in line and "#genre#" not in line:
                        try:
                            url = line.split(',')[1].strip()
                            url_hash = get_url_hash(url)
                            if url_hash not in category['url_set']:
                                category['lines'].append(line)
                                category['url_set'].add(url_hash)
                                added_count += 1
                        except Exception as e:
                            print(f"⚠️ 处理手工频道行错误: {e}, 行: {line}")
                
                print(f"✅ 手工区处理完成: {category['name']} ({added_count}个频道)")
            else:
                print(f"⚠️ 手工区文件不存在: {file_path}")
        
        # 特殊处理：将手工区的地方台频道添加到对应的地方台分类
        for manual_key, manual_category in self.manual_categories.items():
            if manual_key in self.local_categories:
                # 找到对应的地方台分类
                local_category = self.local_categories[manual_key]
                for line in manual_category['lines']:
                    try:
                        url = line.split(',')[1].strip()
                        url_hash = get_url_hash(url)
                        if url_hash not in local_category['url_set']:
                            local_category['manual_lines'].append(line)
                            local_category['url_set'].add(url_hash)
                    except Exception as e:
                        print(f"⚠️ 处理手工地方台频道错误: {e}")
    
    def get_output_lines(self, output_type='full'):
        """获取输出行"""
        output_lines = []
        
        # 主频道
        for key, category in self.main_categories.items():
            if category['lines']:
                sorted_lines = self.sort_data(category['dictionary'], category['lines'])
                output_lines.extend([f"{category['output_name']},#genre#"] + sorted_lines + ['\n'])
        
        # 地方台 - 按名称排序
        local_items = sorted(self.local_categories.items(), key=lambda x: x[1]['name'])
        for key, category in local_items:
            if category['lines'] or category['manual_lines']:
                # 合并自动分发和手工区频道
                all_lines = category['lines'] + category['manual_lines']
                sorted_lines = self.sort_data(category['dictionary'], all_lines)
                output_lines.extend([f"{category['output_name']},#genre#"] + sorted_lines + ['\n'])
        
        # 手工区（独立分类）
        manual_items = sorted(self.manual_categories.items(), key=lambda x: x[1]['name'])
        for key, category in manual_items:
            if category['lines'] and key not in self.local_categories:  # 避免重复显示地方台的手工区
                output_lines.extend([f"{category['output_name']},#genre#"] + sorted(category['lines']) + ['\n'])
        
        # 特殊分类
        for key, category in self.special_categories.items():
            if category['lines']:
                output_lines.extend([f"{category['output_name']},#genre#"] + category['lines'] + ['\n'])
        
        return output_lines
    
    def sort_data(self, order, data):
        """按指定顺序排序数据"""
        try:
            order_dict = {name: i for i, name in enumerate(order)}
            def sort_key(line):
                try:
                    name = line.split(',')[0]
                    return order_dict.get(name, len(order))
                except:
                    return len(order)
            return sorted(data, key=sort_key)
        except Exception as e:
            print(f"⚠️ 排序数据错误: {e}")
            return sorted(data)
    
    def get_statistics(self):
        """获取统计信息"""
        stats = {}
        
        # 主频道统计
        for key, category in self.main_categories.items():
            stats[category['name']] = len(category['lines'])
        
        # 地方台统计
        for key, category in self.local_categories.items():
            stats[category['name']] = len(category['lines']) + len(category['manual_lines'])
        
        # 手工区统计
        for key, category in self.manual_categories.items():
            if key not in self.local_categories:  # 避免重复统计
                stats[category['name']] = len(category['lines'])
        
        # 特殊分类统计
        for key, category in self.special_categories.items():
            stats[category['name']] = len(category['lines'])
        
        return stats

    def normalize_date_to_md(text):
        """日期格式标准化"""
        try:
            text = text.strip()
            def format_md(m):
                month = int(m.group(1))
                day = int(m.group(2))
                after = m.group(3) or ''
                if not after.startswith(' '):
                    after = ' ' + after
                return f"{month}-{day}{after}"

            text = re.sub(r'^0?(\d{1,2})/0?(\d{1,2})(.*)', format_md, text)
            text = re.sub(r'^\d{4}-0?(\d{1,2})-0?(\d{1,2})(.*)', format_md, text)
            text = re.sub(r'^0?(\d{1,2})月0?(\d{1,2})日(.*)', format_md, text)
            return text
        except Exception as e:
            print(f"⚠️ 日期格式化错误: {e}, 文本: {text}")

        if os.path.exists(m3u_file):

        else:

    if all_files_ok:
        print("🎉 所有文件生成成功！")

def process_url(url):
    """处理单个URL源"""
    try:
        print(f"🌐 处理URL: {url}")
        others_lines.append(f"◆◆◆ {url}")
        
        response_text = get_http_response(url)
        if not response_text:
            print(f"❌ 获取URL内容失败: {url}")
            others_lines.append(f"❌ 获取失败: {url}\n")
            return

        # 检查是否为M3U格式
        is_m3u = response_text.startswith("#EXTM3U") or response_text.startswith("#EXTINF")
        if get_url_file_extension(url) in [".m3u", ".m3u8"] or is_m3u:
            response_text = convert_m3u_to_txt(response_text)

        lines = response_text.split('\n')
        valid_lines = 0
        
        for line in lines:
            if ("#genre#" not in line and "," in line and "://" in line and 
                not should_skip_url(line)):
                
                try:
                    channel_name, channel_address = line.split(',', 1)
                    
                    # 处理带#号的加速源
                    if "#" not in channel_address:
                        channel_manager.process_channel_line(line)
                        valid_lines += 1
                    else:
                        url_list = channel_address.split('#')
                        for channel_url in url_list:
                            channel_manager.process_channel_line(f'{channel_name},{channel_url}')
                            valid_lines += 1
                except Exception as e:
                    print(f"⚠️ 处理行错误: {e}, 行: {line}")

        print(f"✅ 处理完成: {valid_lines} 个有效频道")
        others_lines.append(f"✅ 完成: {valid_lines} 个频道\n")

    except Exception as e:
        print(f"❌ 处理URL时发生错误 {url}: {e}")
        others_lines.append(f"❌ 错误: {e}\n")

if __name__ == "__main__":
