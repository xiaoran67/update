核心处理逻辑

```python
import os
from collections import defaultdict

class ChannelManager:
    def __init__(self, base_path):
        self.base_path = base_path
        self.channel_sources = defaultdict(list)  # 存储所有频道的直播源
        
    def load_main_channels(self):
        """加载主频道（只有名称）"""
        main_path = os.path.join(self.base_path, "主频道")
        main_channels = []
        
        for file in os.listdir(main_path):
            if file.endswith('.txt'):
                with open(os.path.join(main_path, file), 'r', encoding='utf-8') as f:
                    for line in f:
                        channel_name = line.strip()
                        if channel_name:
                            main_channels.append(channel_name)
        return main_channels
    
    def load_local_channels(self):
        """加载地方台（只有名称）"""
        local_path = os.path.join(self.base_path, "地方台")
        local_channels = []
        
        for file in os.listdir(local_path):
            if file.endswith('.txt'):
                with open(os.path.join(local_path, file), 'r', encoding='utf-8') as f:
                    for line in f:
                        channel_name = line.strip()
                        if channel_name:
                            local_channels.append(channel_name)
        return local_channels
    
    def load_manual_sources(self):
        """加载手工区（包含直播源）"""
        manual_path = os.path.join(self.base_path, "手工区")
        
        for file in os.listdir(manual_path):
            if file.endswith('.txt'):
                with open(os.path.join(manual_path, file), 'r', encoding='utf-8') as f:
                    for line in f:
                        if ',' in line:
                            channel_name, source = line.strip().split(',', 1)
                            if channel_name and source:
                                # 去重添加
                                if source not in self.channel_sources[channel_name]:
                                    self.channel_sources[channel_name].append(source)
    
    def generate_playlist(self, output_file="output.m3u"):
        """生成最终的播放列表"""
        # 1. 获取所有频道名称（主频道+地方台）
        main_channels = self.load_main_channels()
        local_channels = self.load_local_channels()
        
        # 2. 加载手工区的直播源
        self.load_manual_sources()
        
        # 3. 按字典排序
        all_channels = sorted(set(main_channels + local_channels))
        
        # 4. 生成播放列表
        with open(output_file, 'w', encoding='utf-8') as f:
            f.write('#EXTM3U\n')
            
            for channel in all_channels:
                sources = self.channel_sources.get(channel, [])
                if sources:  # 只在有直播源时才输出
                    for source in sources:
                        f.write(f'#EXTINF:-1,{channel}\n')
                        f.write(f'{source}\n')
        
        print(f"播放列表已生成: {output_file}")
        print(f"包含 {len(all_channels)} 个频道")
```

使用方法

```python
# 简单使用
manager = ChannelManager("/storage/emulated/0/9999...")
manager.generate_playlist()

# 或者指定输出文件
manager.generate_playlist("我的频道列表.m3u")
```

工作流程

1. 扫描主频道 → 获取频道名称列表
2. 扫描地方台 → 获取频道名称列表
3. 扫描手工区 → 建立"频道名→直播源"的映射
4. 合并去重 → 主频道+地方台的所有频道名称
5. 字典排序 → 按频道名称排序
6. 生成列表 → 只输出有直播源的频道

优势

· ✅ 主频道/地方台：纯频道名称，便于维护
· ✅ 手工区：包含完整直播源信息
· ✅ 自动去重：同名频道自动合并
· ✅ 灵活增删：随时添加删除文件即可
· ✅ 字典排序：输出整齐规范
